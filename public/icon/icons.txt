# Icon index
#
# All top-bar icons live in `public/icon/<name>.svg` and are referenced directly
# by URL (`/icon/<name>.svg`, base-aware via `iconSrc()` in `src/icons/index.ts`).
#
# There is no `color/` layer and no monochrome fallback set anymore: an icon is
# just a file in this directory, resolved by its basename from `site.config.ts`.
#
# ---------------------------------------------------------------------------
# NAVIGATION (site.config.ts nav[].icon)
#   home     -> /icon/home.svg
#   archive  -> /icon/archive.svg
#   gallery  -> /icon/gallery.svg
#   friends  -> /icon/friends.svg
#
# LANGUAGE BUTTON
#   globe    -> /icon/globe.svg
#
# 「链接」 PANEL (site.config.ts menu[].icon)
#   link     -> /icon/link.svg
#   github   -> /icon/github.svg
#   email    -> /icon/email.svg
#   cloud    -> /icon/cloud.svg
#   wechat   -> /icon/wechat.svg   (click copies wechatId)
#   rss      -> /icon/rss.svg
#
# ---------------------------------------------------------------------------
# NOT part of the coloured icon set: the search, theme and hamburger buttons
# draw their SVGs inline in their own components and stay `currentColor`, so
# they turn white over a Hero image.
#
# Music widget:
#   neteasecloudmusic | /icon/neteasecloudmusic.svg | referenced by MusicWidget.astro
